cmake_minimum_required(VERSION 3.22)
project(satori_model_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT SATORI_LLAMA_SOURCE)
    message(FATAL_ERROR "Supply the pinned official llama.cpp checkout with SATORI_LLAMA_SOURCE")
endif()
add_subdirectory("${SATORI_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(satori-model-reference reference.cpp)
target_include_directories(satori-model-reference PRIVATE "${SATORI_LLAMA_SOURCE}/vendor")
target_link_libraries(satori-model-reference PRIVATE llama)
